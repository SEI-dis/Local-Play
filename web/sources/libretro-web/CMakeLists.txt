cmake_minimum_required(VERSION 3.20)
project(LocalWebCores C CXX)
if(NOT EMSCRIPTEN OR NOT CORE_ROOT)
 message(FATAL_ERROR "Use emcmake and set CORE_ROOT to the pinned core source directory")
endif()
file(TO_CMAKE_PATH "${CORE_ROOT}" CORE_ROOT)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -O3")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -fexceptions")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static cores" FORCE)
set(CLOWNCD_CHD OFF CACHE BOOL "No CHD library" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_SndFile ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Git ON)
set(CC_USE_C99_INTEGERS ON CACHE BOOL "Fixed width integers" FORCE)
set(FCEU ${CORE_ROOT}/fceumm)
set(COMMON ${FCEU}/src/drivers/libretro/libretro-common)
file(GLOB FCEU_FILES ${FCEU}/src/boards/*.c ${FCEU}/src/input/*.c)
set(FCEU_MAIN cart cheat crc32 fceu-endian fceu-memory fceu fds fds_apu file filter general input md5 nsf palette ppu sound state video vsuni ines unif x6502)
foreach(unit IN LISTS FCEU_MAIN)
 list(APPEND FCEU_FILES ${FCEU}/src/${unit}.c)
endforeach()
list(APPEND FCEU_FILES ${FCEU}/src/drivers/libretro/libretro.c ${FCEU}/src/drivers/libretro/libretro_dipswitch.c)
set(COMMON_FILES compat/compat_posix_string.c compat/compat_snprintf.c compat/compat_strcasestr.c compat/compat_strl.c compat/fopen_utf8.c encodings/encoding_utf.c file/file_path.c file/file_path_io.c streams/file_stream.c streams/file_stream_transforms.c streams/memory_stream.c string/stdstring.c time/rtime.c vfs/vfs_implementation.c)
foreach(unit IN LISTS COMMON_FILES)
 list(APPEND FCEU_FILES ${COMMON}/${unit})
endforeach()
add_library(fceumm STATIC ${FCEU_FILES})
target_include_directories(fceumm PRIVATE ${FCEU}/src ${FCEU}/src/boards ${FCEU}/src/input ${FCEU}/src/drivers/libretro ${COMMON}/include)
target_compile_definitions(fceumm PRIVATE __LIBRETRO__ FCEU_VERSION_NUMERIC=9813 FRONTEND_SUPPORTS_RGB565 LSB_FIRST)
set(MESEN ${CORE_ROOT}/mesen-s)
file(READ ${MESEN}/Libretro/Makefile.common MESEN_MAKE)
string(REGEX MATCHALL "\\$\\([A-Z_]+\\)/[A-Za-z0-9_./-]+\\.(cpp|c)" MESEN_FILES "${MESEN_MAKE}")
string(REPLACE "$(LIBRETRO_DIR)" "${MESEN}/Libretro" MESEN_FILES "${MESEN_FILES}")
string(REPLACE "$(CORE_DIR)" "${MESEN}/Core" MESEN_FILES "${MESEN_FILES}")
string(REPLACE "$(UTIL_DIR)" "${MESEN}/Utilities" MESEN_FILES "${MESEN_FILES}")
string(REPLACE "$(SEVENZIP_DIR)" "${MESEN}/SevenZip" MESEN_FILES "${MESEN_FILES}")
add_library(mesen STATIC ${MESEN_FILES})
target_compile_definitions(mesen PRIVATE LIBRETRO)
add_subdirectory(${CORE_ROOT}/clownmdemu clown EXCLUDE_FROM_ALL)
function(webcore name core extension)
 add_executable(${name} bridge.c)
 target_include_directories(${name} PRIVATE ${COMMON}/include)
 target_compile_definitions(${name} PRIVATE WEB_EXTENSION="${extension}")
 target_link_libraries(${name} PRIVATE ${core})
 set_target_properties(${name} PROPERTIES SUFFIX ".js")
 target_link_options(${name} PRIVATE --no-entry -fexceptions -sMODULARIZE=1 -sEXPORT_NAME=createCore -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=4194304 -sENVIRONMENT=web,node "-sEXPORTED_FUNCTIONS=['_malloc','_free']" "-sEXPORTED_RUNTIME_METHODS=['HEAPU8','HEAP16']")
endfunction()
webcore(nes fceumm nes)
webcore(snes mesen sfc)
webcore(md clownmdemu_libretro md)
